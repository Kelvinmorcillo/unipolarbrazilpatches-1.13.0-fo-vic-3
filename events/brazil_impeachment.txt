namespace = brz_impeachment

brz_impeachment.1 = {
	type = country_event
	placement = ROOT
	title = brz_impeachment.1.t
	desc = brz_impeachment.1.d
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3

	trigger = {
		c:BRZ ?= this
		game_date >= 1992.9.29
		game_date <= 1992.12.29
		NOT = { has_variable = brz_collor_impeachment_started }
	}

	immediate = {
		set_variable = brz_collor_impeachment_started
	}

	option = {
		name = brz_impeachment.1.a
		trigger_event = { id = brz_impeachment.2 days = 3 popup = yes }
	}
}

brz_impeachment.3 = {
	type = country_event
	placement = ROOT
	title = brz_impeachment.3.t
	desc = brz_impeachment.3.d
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3

	trigger = {
		c:BRZ ?= this
		game_date >= 1992.12.29
		game_date <= 1993.12.31
		has_variable = brz_itamar_assumed_presidency
		NOT = { has_variable = brz_collor_impeachment_concluded }
	}

	immediate = {
		set_variable = brz_collor_impeachment_concluded
		set_variable = brz_collor_impeachment_enforced
		character:BRZ_itamar_franco = {
			set_character_as_ruler = yes
		}
		ig:ig_petty_bourgeoisie = {
			leader = {
				exile_character = yes
			}
		}
	}

	option = {
		name = brz_impeachment.3.a
	}
}
