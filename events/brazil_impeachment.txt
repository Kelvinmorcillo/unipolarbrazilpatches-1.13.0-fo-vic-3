namespace = brz_impeachment

brz_impeachment.1 = {
	type = country_event
	placement = ROOT
	title = brz_impeachment.1.t
	desc = brz_impeachment.1.d
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3

	trigger = {
		c:BRZ ?= this
		game_date >= 1992.9.29
		NOT = { has_variable = brz_collor_impeachment_started }
	}

	immediate = {
		set_variable = brz_collor_impeachment_started
	}

	option = {
		name = brz_impeachment.1.a
		trigger_event = { id = brz_impeachment.2 days = 3 popup = yes }
	}
}

brz_impeachment.5 = {
	type = country_event
	placement = ROOT
	title = brz_impeachment.5.t
	desc = brz_impeachment.5.d
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3

	trigger = {
		c:BRZ ?= this
		game_date >= 2016.5.12
		has_variable = brz_dilma_impeachment_started
		NOT = { has_variable = brz_bolsonaro_armed_forces_leader }
	}

	immediate = {
		set_variable = brz_bolsonaro_armed_forces_leader
		create_character = {
			template = brz_jair_bolsonaro_character_template
			on_created = {
				set_as_interest_group_leader = yes
			}
		}
		# Dilma is suspended: Temer takes over as acting president
		brz_temer_takes_office_effect = yes
	}

	option = {
		name = brz_impeachment.5.a
		trigger_event = { id = brz_impeachment.6 days = 111 popup = yes } # conviction - 31 Aug 2016
	}
}
