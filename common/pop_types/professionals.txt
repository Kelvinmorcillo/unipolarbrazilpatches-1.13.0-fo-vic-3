professionals = {
	texture = "gfx/interface/icons/pops_icons/professionals.dds"
	start_quality_of_life = 12
	color = hsv{ 0.00 0.00 1.00 }
	wage_weight = 22
	paid_private_wage = yes
	literacy_target = 0.16
	education_access = 0.6
	dependent_wage = 0.5	# per year
	unemployment = yes

	# 40% - 90% politically engaged -- better educated and better paid than
	# Associates, so more likely to organise
	political_engagement_base = 0.40
	political_engagement_literacy_factor = 0.5

	political_engagement_mult = {
		value = 1

		add = {
			desc = "POP_STARVATION"

			if = {
				limit = {
					is_in_mild_starvation = yes
				}
				value = 0.5
			}
			else_if = {
				limit = {
					is_in_severe_starvation = yes
				}
				value = 1.5
			}
		}

		multiply = {
			desc = "HAS_POLITICAL_AGITATION"
			value = 1.5
		}
	}

	qualifications_growth_desc = "PROFESSIONALS_QUALIFICATIONS_DESC"
	qualifications = {
		# baseline: ( literacy - 0.06 ) * 130 + ( wealth / 15 )
		# Literacy here means COLLEGE EDUCATED, so a developed nation sits near
		# 0.20 and China near 0.08. Threshold 0.06 sits above the Associate's
		# 0.025, giving 18.2 at a developed nation against their 42.0 -- scarce
		# but obtainable -- and 2.6 in China, a 7x spread.
		add = {
			value = literacy_rate
			subtract = 0.06
			min = 0
			multiply = 130
			desc = "QUALIFICATIONS_LITERACY_FACTOR"
		}
		add = {
			value = wealth
			divide = 15
			desc = "QUALIFICATIONS_WEALTH_FACTOR"
		}
		if = {
			limit = {
				is_pop_type = clerks
			}
			multiply = {
				value = 2
				desc = "QUALIFICATIONS_FAVORED_TYPE"
			}
		}
		if = {
			limit = {
				is_peasant_under_serfdom = yes
			}
			multiply = {
				value = 0.1
				desc = "QUALIFICATIONS_PEASANTS_UNDER_SERFDOM"
			}
		}
	}

	portrait_age = {
		integer_range = {
			min = define:NPortrait|GRACEFUL_AGING_START
			max = define:NPortrait|GRACEFUL_AGING_END
		}
	}
	portrait_pose = { value = 0 }
	portrait_is_female = { always = yes }
}
